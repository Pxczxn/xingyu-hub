## 一、性能问题

### 1.1 重复的 API 请求（高优先级）

**问题描述**：多个页面在组件挂载时都会独立请求相同的数据，没有缓存机制。

**具体实例**：

```typescript
// HomePage.tsx - 请求热门话题
topicsApi.getTrendingTopics(4)

// DiscoverPage.tsx - 同样请求热门话题
topicsApi.getTopics()

// TopicsPage.tsx - 再次请求相同数据
topicsApi.getTopics()
```

**影响**：
- 用户在首页 → 发现 → 话题页之间切换时，每次都重新请求相同数据
- 增加服务器负载
- 浪费用户流量

**建议方案**：
```typescript
// 方案 1: 引入 React Query / TanStack Query
import { useQuery } from '@tanstack/react-query';

export function useTopics() {
  return useQuery({
    queryKey: ['topics'],
    queryFn: () => topicsApi.getTopics(),
    staleTime: 5 * 60 * 1000, // 5 分钟内复用缓存
  });
}

// 方案 2: 简单的内存缓存层
// src/lib/cache.ts
const cache = new Map<string, { data: unknown; expires: number }>();

export function cachedFetch<T>(
  key: string,
  fetcher: () => Promise<T>,
  ttl = 300000 // 5分钟
): Promise<T> {
  const cached = cache.get(key);
  if (cached && Date.now() < cached.expires) {
    return Promise.resolve(cached.data as T);
  }
  
  return fetcher().then(data => {
    cache.set(key, { data, expires: Date.now() + ttl });
    return data;
  });
}
```

**优先级**：⭐⭐⭐⭐ 高  
**预计收益**：减少 30-50% 的 API 请求量

### 1.3 未优化的 useEffect 依赖（中优先级）

**问题描述**：多个页面的 `useEffect` 使用了自增 key 来触发重载，这是一个反模式。

**具体实例**：

```typescript
// TopicsPage.tsx, GalaxyListPage.tsx, EventsPage.tsx 等多处
const [reloadKey, setReloadKey] = useState(0);

useEffect(() => {
  // ... 加载数据
}, [reloadKey]); // ❌ 依赖于人工触发的 key

// 在错误处理中
<Button onClick={() => setReloadKey((key) => key + 1)}>
  重新加载
</Button>
```

**问题**：
- 这种模式让数据流不透明
- 难以追踪数据何时刷新
- 无法利用 React 的自动优化

**建议方案**：
```typescript
// ✅ 更好的模式：显式的 refetch 函数
function useFetchTopics() {
  const [state, setState] = useState<LoadState>({ kind: 'loading' });
  
  const fetch = useCallback(async () => {
    setState({ kind: 'loading' });
    try {
      const data = await topicsApi.getTopics();
      setState({ kind: 'ready', data });
    } catch (error) {
      setState({ kind: 'error', error });
    }
  }, []);
  
  useEffect(() => {
    fetch();
  }, []); // 只在挂载时加载
  
  return { ...state, refetch: fetch }; // 返回显式的重载函数
}

// 使用时
const { data, refetch } = useFetchTopics();
<Button onClick={refetch}>重新加载</Button>
```

**优先级**：⭐⭐⭐ 中  
**预计收益**：提高代码可维护性，减少意外的重渲染

```typescript
// src/lib/error-messages.ts
export function getErrorMessage(error: unknown): {
  title: string;
  description: string;
  action?: string;
} {
  if (error instanceof ApiError) {
    switch (error.problem.status) {
      case 401:
        return {
          title: '需要登录',
          description: '请登录后继续操作',
          action: '前往登录',
        };
      case 403:
        return {
          title: '没有权限',
          description: '你无权访问此内容',
          action: '返回首页',
        };
      case 404:
        return {
          title: '内容不存在',
          description: '该内容可能已被删除或移动',
          action: '返回上一页',
        };
      case 429:
        return {
          title: '操作太频繁',
          description: '请稍后再试',
          action: undefined,
        };
      case 500:
        return {
          title: '服务器错误',
          description: '我们正在处理这个问题',
          action: '重新加载',
        };
      default:
        return {
          title: '请求失败',
          description: error.problem.detail || '未知错误',
          action: '重试',
        };
    }
  }
  
  if (error instanceof Error) {
    return {
      title: '操作失败',
      description: error.message,
      action: '重试',
    };
  }
  
  return {
    title: '发生错误',
    description: '请稍后重试',
    action: '重试',
  };
}

// 使用时
catch (error) {
  const { title, description, action } = getErrorMessage(error);
  setState({ kind: 'error', title, description, action });
}
```

**优先级**：⭐⭐⭐ 中  
**预计收益**：减少用户困惑，提高问题解决效率

## 四、代码质量问题

### 4.1 重复的状态管理模式（中优先级）

**问题描述**：至少 15+ 个页面使用完全相同的加载/错误/成功状态模式，但每个都重新实现。

**具体实例**：

```typescript
// 以下模式在多个文件中重复出现：
// - MomentsPage.tsx
// - TopicsPage.tsx
// - GalaxyListPage.tsx
// - EventsPage.tsx
// - CollectionsPage.tsx
// - ...

type LoadState = 'loading' | 'error' | 'ready';
const [loadState, setLoadState] = useState<LoadState>('loading');
const [items, setItems] = useState<ItemType[]>([]);

useEffect(() => {
  let active = true;
  setLoadState('loading');
  
  api.fetchItems()
    .then(data => {
      if (active) {
        setItems(data);
        setLoadState('ready');
      }
    })
    .catch(() => {
      if (active) setLoadState('error');
    });
    
  return () => { active = false; };
}, []);
```

**建议方案**：
```typescript
// src/hooks/useAsyncData.ts
export function useAsyncData<T>(
  fetcher: () => Promise<T>,
  deps: React.DependencyList = []
) {
  type State = 
    | { status: 'loading' }
    | { status: 'error'; error: Error }
    | { status: 'success'; data: T };
    
  const [state, setState] = useState<State>({ status: 'loading' });
  
  const refetch = useCallback(async () => {
    setState({ status: 'loading' });
    try {
      const data = await fetcher();
      setState({ status: 'success', data });
    } catch (error) {
      setState({ 
        status: 'error', 
        error: error instanceof Error ? error : new Error('Unknown error')
      });
    }
  }, [fetcher]);
  
  useEffect(() => {
    refetch();
  }, deps);
  
  return { ...state, refetch };
}

// 使用时
function TopicsPage() {
  const { status, data, error, refetch } = useAsyncData(
    () => topicsApi.getTopics(),
    []
  );
  
  if (status === 'loading') return <PageState kind="loading" />;
  if (status === 'error') return (
    <PageState 
      kind="error" 
      action={<Button onClick={refetch}>重试</Button>}
    />
  );
  
  return <TopicList topics={data} />;
}
```

**优先级**：⭐⭐⭐ 中  
**预计收益**：减少 500+ 行重复代码，统一错误处理逻辑

## 五、架构建议

### 5.1 数据获取层统一（建议）

**当前状况**：每个页面直接调用 API，没有中间层。

**建议架构**：

```
当前：Component → API Client → Backend
               ↓
建议：Component → Query Hook → Cache → API Client → Backend
```

**实现方案**：
```typescript
// src/features/topics/queries.ts
export const topicQueries = {
  all: () => ['topics'] as const,
  list: () => [...topicQueries.all(), 'list'] as const,
  detail: (slug: string) => [...topicQueries.all(), 'detail', slug] as const,
  creators: (slug: string) => [...topicQueries.detail(slug), 'creators'] as const,
};

export function useTopics() {
  return useQuery({
    queryKey: topicQueries.list(),
    queryFn: () => topicsApi.getTopics(),
    staleTime: 5 * 60 * 1000,
  });
}

export function useTopic(slug: string) {
  return useQuery({
    queryKey: topicQueries.detail(slug),
    queryFn: () => topicsApi.getTopicDetail(slug),
    staleTime: 2 * 60 * 1000,
  });
}

export function useTopicCreators(slug: string) {
  return useQuery({
    queryKey: topicQueries.creators(slug),
    queryFn: () => topicsApi.getTopicCreators(slug),
    staleTime: 10 * 60 * 1000,
  });
}
```

**优势**：
- 自动缓存和重用数据
- 自动处理加载/错误状态
- 支持乐观更新
- 支持预取（prefetch）
- DevTools 支持

**工作量评估**：2-3 周（逐步迁移）

### 5.2 表单处理统一（建议）

**建议使用**：React Hook Form

```typescript
// src/features/auth/LoginForm.tsx
import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import * as z from 'zod';

const loginSchema = z.object({
  login: z.string().min(1, '请输入用户名或邮箱'),
  password: z.string().min(6, '密码至少 6 位'),
});

type LoginForm = z.infer<typeof loginSchema>;

export function LoginForm() {
  const {
    register,
    handleSubmit,
    formState: { errors, isSubmitting },
  } = useForm<LoginForm>({
    resolver: zodResolver(loginSchema),
  });
  
  const onSubmit = async (data: LoginForm) => {
    await authApi.login(data);
  };
  
  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      <input {...register('login')} />
      {errors.login && <span>{errors.login.message}</span>}
      
      <input type="password" {...register('password')} />
      {errors.password && <span>{errors.password.message}</span>}
      
      <button type="submit" disabled={isSubmitting}>
        登录
      </button>
    </form>
  );
}
```
